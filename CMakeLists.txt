cmake_minimum_required(VERSION 3.16)
project(exfat_shrink C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(EXFAT_SOURCES main.c exfat.c exfat.h verify.c verify.h shrink.c metadata.c patch.c patch.h sha256.c sha256.h file_hash.c file_hash.h live.c live.h recovery_state.c recovery_state.h recovery_sector.c recovery_sector.h)
add_executable(exfat_shrink ${EXFAT_SOURCES})
if(MSVC)
  target_compile_options(exfat_shrink PRIVATE /W4)
else()
  target_compile_options(exfat_shrink PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  if(WIN32)
    add_executable(exfat_fault_test ${EXFAT_SOURCES})
    target_compile_definitions(exfat_fault_test PRIVATE EXFAT_TEST_FAULTS=1)
  endif()
  add_executable(test_recovery_sector test_recovery_sector.c recovery_sector.c)
  add_test(NAME torn_sector_validation COMMAND test_recovery_sector)
  add_executable(test_recovery_state test_recovery_state.c recovery_state.c)
  add_test(NAME recovery_state_validation COMMAND test_recovery_state)
  add_test(NAME large_sparse_geometry COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_large_geometry.py $<TARGET_FILE:exfat_shrink>)
  add_test(NAME cli_endurance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_endurance.py $<TARGET_FILE:exfat_shrink>)
  add_test(NAME extended_coverage COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_extended.py $<TARGET_FILE:exfat_shrink>)
  add_executable(test_patch test_patch.c exfat.c verify.c metadata.c patch.c sha256.c file_hash.c)
  add_test(NAME metadata_transactions COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_patch.py $<TARGET_FILE:exfat_shrink> $<TARGET_FILE:test_patch>)
  add_executable(test_sha256 test_sha256.c sha256.c)
  add_test(NAME hash_safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_hashes.py $<TARGET_FILE:exfat_shrink> $<TARGET_FILE:test_sha256>)
  add_test(NAME image_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_images.py $<TARGET_FILE:exfat_shrink>)
endif()
